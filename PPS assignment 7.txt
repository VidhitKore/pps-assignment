#include <stdio.h>

#define MAX_STUDENTS 50
#define SUBJECTS 3

struct Student {
    int rollNo;
    char name[50];
    float marks[SUBJECTS];
    float total;
    float percentage;
    char grade;
};

char calculateGrade(float percentage) {
    if (percentage >= 85.0f) return 'A';
    if (percentage >= 70.0f) return 'B';
    if (percentage >= 50.0f) return 'C';
    if (percentage >= 35.0f) return 'D';
    return 'F';
}

int main() {
    int n, i, j;
    struct Student s[MAX_STUDENTS];

    printf("Enter number of students (1-%d): ", MAX_STUDENTS);
    scanf("%d", &n);

    if (n <= 0 || n > MAX_STUDENTS) {
        printf("Invalid number of students.\n");
        return 1;
    }

    for (i = 0; i < n; i++) {
        printf("\n--- Enter Details for Student %d ---\n", i + 1);
        printf("Roll Number: ");
        scanf("%d", &s[i].rollNo);

        printf("Name: ");
        scanf(" %49[^\n]", s[i].name);

        s[i].total = 0.0f;
        for (j = 0; j < SUBJECTS; j++) {
            printf("Enter marks for Subject %d (out of 100): ", j + 1);
            scanf("%f", &s[i].marks[j]);
            s[i].total += s[i].marks[j];
        }

        s[i].percentage = s[i].total / SUBJECTS;
        s[i].grade = calculateGrade(s[i].percentage);
    }

    printf("\n==================== STUDENT RESULT SHEET ====================\n");
    printf("Roll No\tName\t\tTotal\tPercentage\tGrade\n");
    printf("--------------------------------------------------------------\n");
    for (i = 0; i < n; i++) {
        printf("%d\t%-15s\t%.2f\t%.2f%%\t\t%c\n",
               s[i].rollNo, s[i].name, s[i].total, s[i].percentage, s[i].grade);
    }
    printf("==============================================================\n");

    return 0;
}
