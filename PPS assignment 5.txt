#include <stdio.h>
#include <string.h>

#define MAX 100

int main() {
    int choice, cmp;
    char str1[MAX], str2[MAX], temp[MAX];

    printf("Enter first string: ");
    fgets(str1, MAX, stdin);
    str1[strcspn(str1, "\n")] = '\0'; // Remove trailing newline

    printf("\n--- String Operations Menu ---\n");
    printf("1. String Length (strlen)\n");
    printf("2. String Copy (strcpy)\n");
    printf("3. String Concatenate (strcat)\n");
    printf("4. String Compare (strcmp)\n");
    printf("5. Find Character (strchr)\n");
    printf("6. Find Substring (strstr)\n");
    printf("Enter your choice (1-6): ");
    scanf("%d", &choice);
    getchar(); // Consume leftover newline

    switch (choice) {
        case 1:
            printf("Length of '%s' = %zu\n", str1, strlen(str1));
            break;

        case 2:
            strcpy(temp, str1);
            printf("Copied string = %s\n", temp);
            break;

        case 3:
            printf("Enter second string to concatenate: ");
            fgets(str2, MAX, stdin);
            str2[strcspn(str2, "\n")] = '\0';

            strcpy(temp, str1);
            strcat(temp, str2);
            printf("Concatenated string = %s\n", temp);
            break;

        case 4:
            printf("Enter second string to compare: ");
            fgets(str2, MAX, stdin);
            str2[strcspn(str2, "\n")] = '\0';

            cmp = strcmp(str1, str2);
            if (cmp == 0) {
                printf("Both strings are equal.\n");
            } else if (cmp < 0) {
                printf("'%s' comes before '%s' lexicographically.\n", str1, str2);
            } else {
                printf("'%s' comes after '%s' lexicographically.\n", str1, str2);
            }
            break;

        case 5: {
            char ch;
            printf("Enter character to search: ");
            scanf("%c", &ch);

            char *pos = strchr(str1, ch);
            if (pos != NULL) {
                printf("Character '%c' found at index %ld.\n", ch, pos - str1);
            } else {
                printf("Character '%c' not found.\n", ch);
            }
            break;
        }

        case 6: {
            printf("Enter substring to search: ");
            fgets(str2, MAX, stdin);
            str2[strcspn(str2, "\n")] = '\0';

            char *sub = strstr(str1, str2);
            if (sub != NULL) {
                printf("Substring '%s' found at index %ld.\n", str2, sub - str1);
            } else {
                printf("Substring '%s' not found.\n", str2);
            }
            break;
        }

        default:
            printf("Invalid choice! Please enter a number between 1 and 6.\n");
            break;
    }

    return 0;
}
